import pandas as pd
import os

def clean_dataset(input_file, output_file="clean_data.csv", critical_column="CustomerID"):
    """
    Official Data Cleaning Script
    - Removes duplicate rows
    - Removes rows with missing critical column
    - Generates a cleaning report
    """
    try:
        # Check if file exists
        if not os.path.exists(input_file):
            print(f"Error: File '{input_file}' nahi mili.")
            return

        # 1. Load data
        df = pd.read_csv(input_file)
        initial_count = len(df)

        print("--- DATA CLEANING REPORT ---")
        print(f"Initial Row Count: {initial_count}")

        # 2. Remove duplicates
        duplicates_count = df.duplicated().sum()
        df = df.drop_duplicates()
        print(f"Duplicate Rows Found & Removed: {duplicates_count}")

        # 3. Handle missing critical column
        if critical_column in df.columns:
            missing_count = df[critical_column].isna().sum()
            df = df.dropna(subset=[critical_column])
            print(f"Rows with Missing '{critical_column}' Removed: {missing_count}")
        else:
            print(f"Warning: Column '{critical_column}' not found. Skipping NA removal.")

        # 4. Final summary
        final_count = len(df)
        total_dropped = initial_count - final_count

        print(f"Total Rows Dropped: {total_dropped}")
        print(f"Final Clean Row Count: {final_count}")
        print("----------------------------")

        # 5. Save cleaned data
        df.to_csv(output_file, index=False)
        print(f"SUCCESS: Cleaned data saved as '{output_file}'")

    except Exception as e:
        print(f"An error occurred: {e}")
